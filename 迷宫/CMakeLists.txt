cmake_minimum_required(VERSION 3.10)
project(scout_maze)
set(CMAKE_CXX_STANDARD 14)
set(LAB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../实验三扩展/实验二/lab26/catkin_ws/src/basic_control" CACHE PATH "Shared controller sources")
set(LAB3 "${CMAKE_CURRENT_SOURCE_DIR}/../实验三扩展/实验三" CACHE PATH "Experiment 3 navigation sources")
set(FIGURE8 "${CMAKE_CURRENT_SOURCE_DIR}/../实验二扩展/include" CACHE PATH "Figure eight headers")
find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs sensor_msgs nav_msgs std_msgs)
add_executable(maze_controller "${LAB_ROOT}/src/basic_control.cpp" "${LAB3}/src/control_core.cpp" "${LAB3}/src/navigation.cpp")
add_executable(test_laser_maze test/test_laser_maze.cpp "${LAB3}/src/control_core.cpp" "${LAB3}/src/navigation.cpp")
add_executable(test_maze_exit test/test_maze_exit.cpp "${LAB3}/src/control_core.cpp" "${LAB3}/src/navigation.cpp")
foreach(target maze_controller test_laser_maze test_maze_exit)
  target_include_directories(${target} PRIVATE include "${LAB_ROOT}/include" "${FIGURE8}" "${LAB3}/include" ${catkin_INCLUDE_DIRS})
endforeach()
target_link_libraries(maze_controller ${catkin_LIBRARIES})
enable_testing()
foreach(seed 0 5 6 18 24)
  add_test(NAME maze_cones_exit_${seed} COMMAND test_maze_exit ${seed} cones)
endforeach()
add_test(NAME maze_wall_exit COMMAND test_maze_exit 0 walls)
add_test(NAME maze_large_exit COMMAND test_maze_exit 0 large)
add_test(NAME maze_pocket_turn COMMAND test_maze_exit 12 walls)
add_test(NAME maze_pocket_turn_jitter COMMAND test_maze_exit 12 turn_jitter)
add_test(NAME laser_maze COMMAND test_laser_maze)
add_test(NAME maze_interrupted_turn COMMAND test_laser_maze --interrupted-turn)
foreach(variant RANGE 0 7)
  foreach(scene RANGE 0 2)
    add_test(NAME maze_scene_${scene}_variant_${variant} COMMAND test_laser_maze ${variant} ${scene})
  endforeach()
endforeach()
