cmake_minimum_required(VERSION 3.0.2)
project(basic_control)
add_compile_options(-std=c++14 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs sensor_msgs nav_msgs std_msgs)
catkin_package()
include_directories(include ${catkin_INCLUDE_DIRS})
find_path(FIGURE8_INCLUDE_DIR figure_eight.h HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../实验二扩展/include" "/home/nvidia/lab2-extension/include")
if(NOT FIGURE8_INCLUDE_DIR)
  message(FATAL_ERROR "Missing experiment 2 extension figure_eight.h")
endif()
include_directories(${FIGURE8_INCLUDE_DIR})
find_path(LAB3_ROOT src/control_core.cpp HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../实验三" "/home/nvidia/lab3")
if(NOT LAB3_ROOT)
  message(FATAL_ERROR "Missing experiment 3 control_core.cpp")
endif()
include_directories("${LAB3_ROOT}/include")
add_executable(basicControl src/basic_control.cpp "${LAB3_ROOT}/src/control_core.cpp" "${LAB3_ROOT}/src/navigation.cpp")
target_link_libraries(basicControl ${catkin_LIBRARIES})
add_executable(test_motion test/test_motion.cpp "${LAB3_ROOT}/src/control_core.cpp" "${LAB3_ROOT}/src/navigation.cpp")
if(EXISTS "${LAB3_ROOT}/tests/ros_navigation.cpp")
  add_executable(test_navigation_ros "${LAB3_ROOT}/tests/ros_navigation.cpp")
  target_link_libraries(test_navigation_ros ${catkin_LIBRARIES})
endif()
if(EXISTS "${LAB3_ROOT}/tests/live_shadow.cpp")
  add_executable(test_navigation_shadow "${LAB3_ROOT}/tests/live_shadow.cpp" "${LAB3_ROOT}/src/navigation.cpp")
  target_link_libraries(test_navigation_shadow ${catkin_LIBRARIES})
endif()
install(TARGETS basicControl RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
